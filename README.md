/**
 * Script de Integración Básica para Google RCS Business Messaging (RBM)
 * 
 * Ejemplo de implementación backend para enviar un mensaje enriquecido (RCS)
 * utilizando la API de Google Business Communications.
 * 
 * Requisitos:
 * 1. Tener un proyecto en Google Cloud Platform (GCP).
 * 2. Habilitar la 'RCS Business Messaging API' en la consola de GCP.
 * 3. Descargar una cuenta de servicio en formato JSON (Service Account Key) y configurar
 *    la variable de entorno GOOGLE_APPLICATION_CREDENTIALS apuntando a dicho archivo.
 */
const { GoogleAuth } = require('google-auth-library');const axios = require('axios');
// 1. Configuración de autenticación y endpoints oficiales de Googleconst OAUTH_SCOPE = 'https://www.googleapis.com/auth/businesscommunications';const RBM_API_BASE_URL = 'https://rcsbusinessmessaging.googleapis.com/v1';


// Inicializar cliente de autenticación con las credenciales por defecto del entornoconst auth = new GoogleAuth({
  scopes: OAUTH_SCOPE,
});
/**
 * Función principal para enviar una tarjeta enriquecida (Rich Card) a un usuario.
 * @param {string} phoneNumber Número de teléfono en formato E.164 (ej: +521234567890)
 * @param {string} agentId Identificador único de tu Agente RCS verificado
 */async function sendRCSVerifyCard(phoneNumber, agentId) {
  try {
    // Obtener el cliente autorizado y generar el token de acceso OAuth2 de forma dinámica
    const authClient = await auth.getClient();
    const tokenResponse = await authClient.getAccessToken();
    const accessToken = tokenResponse.token;

    // Construcción del recurso del mensaje (Agent Message)
    // Define una tarjeta enriquecida con texto, imagen de la marca y botones interactivos

    const rcsMessagePayload = {
      messageId: `msg_${Date.now()}`, // ID único para rastreo y webhooks de entrega
      contentMessage: {
        richCard: {
          standaloneCard: {
            cardOrientation: 'VERTICAL',
            thumbnailImageAlignment: 'TOP',
            cardContent: {
              title: 'Autenticación Protegida',
              description: 'Tu cuenta de Google ha detectado una solicitud de acceso. Por seguridad, confirma tu identidad presionando el botón de abajo o rechaza la solicitud inmediatamente si no fuiste tú.',
              media: {
                fileUri: 'https://placehold.co/600x400/007bff/ffffff?text=Seguridad+Google',
                thumbnailUri: 'https://placehold.co/150x150/007bff/ffffff?text=Seguridad',
                height: 'MEDIUM'
              },
              suggestions: [
                {
                  reply: {
                    text: 'Sí, soy yo',

                    postbackData: 'auth_approve_click'
                  }
                },
                {
                  reply: {
                    text: 'No, bloquear acceso',
                    postbackData: 'auth_deny_click'
                  }
                }
              ]
            }
          }
        }
      }
    };

    // Estructura de la URL oficial: v1/phones/{phoneNumber}/agentMessages
    const url = `${RBM_API_BASE_URL}/phones/${encodeURIComponent(phoneNumber)}/agentMessages`;


    console.log(`Iniciando envío de mensaje RCS al número: ${phoneNumber}`);

    // Realizar la petición HTTP POST utilizando Axios e inyectar las credenciales Bearer
    const response = await axios.post(url, rcsMessagePayload, {
      headers: {
        'Authorization': `Bearer ${accessToken}`,
        'Content-Type': 'application/json'
      },
      params: {
        agentId: agentId // Identificador del agente requerido por el endpoint de Google
      }
    });

    console.log('Mensaje RCS procesado y enviado con éxito por la plataforma RBM.');
    console.log('ID de Respuesta de Google:', response.data);
    
    return response.data;

  } catch (error) {

    console.error('Error crítico al enviar el mensaje RCS:');
    
    if (error.response) {
      // La API de Google respondió con un código de estado fuera del rango 2xx
      console.error(`Código de estado HTTP devuelto: ${error.response.status}`);
      console.error('Detalles del error de la API RBM:', JSON.stringify(error.response.data, null, 2));
    } else if (error.request) {
      // La petición fue hecha pero no se recibió respuesta (problemas de red)
      console.error('No se recibió respuesta del servidor de Google Business Messaging.');
    } else {
      // Errores internos de código o configuración de autenticación
      console.error('Mensaje de error interno:', error.message);
    }
    throw error;
  }
}

 * @param {string} agentId Identificador único de tu Agente RCS verificado
